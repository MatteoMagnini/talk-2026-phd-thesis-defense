\begin{frame}[c]{Symbolic Knowledge}
    \begin{exampleblock}{Definition}
        Symbolic knowledge is a type of \ac{KR} that uses \emph{symbols} to represent concepts, objects, and relationships in a structured and interpretable way.
        %
        Symbolic \ac{KR} approaches must:
        \begin{itemize}
            \item involve a set of symbols;
            \item symbols can be \emph{combined} following a set of rules;
            \item symbols and combinations of symbols can be \emph{assigned a meaning}.
        \end{itemize}
    \end{exampleblock}

    \begin{alertblock}{Applications}
        \begin{multicols}{3}
            {\small
                \begin{itemize}
                    \item Reasoning
                    \item Logic programming
                    \item Planning
                \end{itemize}
            }
        \end{multicols}
    \end{alertblock}

    \begin{block}{Limits}
        \begin{multicols}{3}
            {\small
                \begin{itemize}
                    \item Brittle \& inflexible
                    \item Acquire \& maintain
                    \item Slow
                \end{itemize}
            }
        \end{multicols}
    \end{block}

\end{frame}
%\\\\\\\\\\\\\\\\\\\\\

%\\\\\\\\\\\\\\\\\\\\\
\begin{frame}[c]{Sub-symbolic Approaches}
    \begin{exampleblock}{Definition}
        Sub-symbolic knowledge is a type of \ac{KR} that represents information in a \emph{distributed} and implicit manner,
        often using \emph{numerical values} or \emph{vectors} to encode concepts and relationships.
        %
        Sub-symbolic \ac{AI} methods include predictive models, based on \emph{statistical patterns} and associations in \emph{data}, rather than explicit rules or symbols.
    \end{exampleblock}

    \begin{alertblock}{Applications}
        \begin{multicols}{2}
            {\small
                \begin{itemize}
                    \item Classification
                    \item Regression
                    \item Embeddings
                    \item Generative \ac{AI}
                \end{itemize}
            }
        \end{multicols}
    \end{alertblock}

    \begin{block}{Limits}
        \begin{multicols}{3}
            {\small
                \begin{itemize}
                    \item Black box
                    \item Outliers
                    \item Hallucinations
                \end{itemize}
            }
        \end{multicols}
    \end{block}

\end{frame}

\begin{frame}[c]{Thesis Goal}
  \begin{alertblock}{Thesis Goal}
    Develop methodologies and frameworks to engineer intelligent \ac{AI} systems,
    focusing on \emph{\ac{SKI}} and \emph{\ac{SKE}} as foundations for human-like, autonomous, and interpretable artificial intelligence.
    %
    This thesis aims to bridge symbolic and sub-symbolic paradigms, enabling \ac{AI} systems to learn, reason, and adapt transparently and robustly.
  \end{alertblock}

  \begin{exampleblock}{Multi-Faceted Scientific Problem}
    \begin{itemize}
      \item \textbf{How} to \underline{inject} and \underline{extract} symbolic knowledge in sub-symbolic models (methods, taxonomies, tools)
      \item \textbf{How} to \emph{measure} the impact of \ac{SKI} and \ac{SKE} on accuracy, \underline{robustness}, and \underline{interpretability} (evaluation metrics)
      \item \textbf{How} to design \ac{NeSy} \ac{AI} systems leveraging \ac{SKI}/\ac{SKE} in practical, high-stakes domains (software frameworks \& real-world deployment)
      \item \textbf{How} to mitigate social and technical risks, e.g., bias, using \ac{SKI} techniques (trustworthiness \& ethics)
    \end{itemize}
  \end{exampleblock}
\end{frame}

\begin{frame}{SKE landscape -- How to extract: Decomposional vs. Pedagogical SKE}

    \begin{exampleblock}{Decompositional SKE Definition}
        if the method \emph{needs} to inspect (even partially) the internal parameters of the underlying black-box predictor,
        e.g., neuron biases or connection weights for \acp{NN}, or support vectors for \acp{SVM}.
    \end{exampleblock}

    \begin{exampleblock}{Pedagogical SKE Definition}
        if the algorithm \emph{does not need} to take into account any internal parameter,
        but it can extract symbolic knowledge by only relying on the predictor's outputs.
    \end{exampleblock}

\end{frame}
