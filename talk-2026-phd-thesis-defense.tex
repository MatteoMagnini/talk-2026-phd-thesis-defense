\begin{frame}[c]{Symbolic Knowledge}
    \begin{exampleblock}{Definition}
        Symbolic knowledge is a type of \ac{KR} that uses \emph{symbols} to represent concepts, objects, and relationships in a structured and interpretable way.
        %
        Symbolic \ac{KR} approaches must:
        \begin{itemize}
            \item involve a set of symbols;
            \item symbols can be combined following a set of rules;
            \item symbols and combinations of symbols can be assigned a meaning.
        \end{itemize}
    \end{exampleblock}

    \begin{alertblock}{Applications}
        \begin{multicols}{3}
            {\small
                \begin{itemize}
                    \item Reasoning
                    \item Logic programming
                    \item Planning
                \end{itemize}
            }
        \end{multicols}
    \end{alertblock}

    \begin{block}{Limits}
        \begin{multicols}{3}
            {\small
                \begin{itemize}
                    \item Brittle \& inflexible
                    \item Acquire \& maintain
                    \item Slow
                \end{itemize}
            }
        \end{multicols}
    \end{block}

\end{frame}
%\\\\\\\\\\\\\\\\\\\\\

%\\\\\\\\\\\\\\\\\\\\\
\begin{frame}[c]{Sub-symbolic Approaches}
    \begin{exampleblock}{Definition}
        Sub-symbolic knowledge is a type of \ac{KR} that represents information in a \emph{distributed} and implicit manner,
        often using \emph{numerical values} or \emph{vectors} to encode concepts and relationships.
        %
        Sub-symbolic \ac{AI} methods include predictive models, based on \emph{statistical patterns} and associations in \emph{data}, rather than explicit rules or symbols.
    \end{exampleblock}

    \begin{alertblock}{Applications}
        \begin{multicols}{2}
            {\small
                \begin{itemize}
                    \item Classification
                    \item Regression
                    \item Embeddings
                    \item Generative \ac{AI}
                \end{itemize}
            }
        \end{multicols}
    \end{alertblock}

    \begin{block}{Limits}
        \begin{multicols}{3}
            {\small
                \begin{itemize}
                    \item Black box
                    \item Outliers
                    \item Hallucinations
                \end{itemize}
            }
        \end{multicols}
    \end{block}

\end{frame}
